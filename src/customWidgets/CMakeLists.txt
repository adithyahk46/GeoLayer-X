set(CUSTOMWIDGETS_TARGET_NAME customWidgets)

add_library(${CUSTOMWIDGETS_TARGET_NAME} STATIC
    colorpickercheckbox.cpp
    colorpickercheckbox.h
)

generate_export_header(${CUSTOMWIDGETS_TARGET_NAME}
    BASE_NAME CUSTOMWIDGETS
    EXPORT_FILE_NAME "${CMAKE_CURRENT_BINARY_DIR}/customWidgets_global.h"
    STATIC_DEFINE CUSTOMWIDGETS_STATIC_DEFINE
)

include(GNUInstallDirs)

target_compile_definitions(${CUSTOMWIDGETS_TARGET_NAME}
    PUBLIC CUSTOMWIDGETS_STATIC_DEFINE
)

target_include_directories(${CUSTOMWIDGETS_TARGET_NAME}
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
        $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
)

target_link_libraries(${CUSTOMWIDGETS_TARGET_NAME}
    PUBLIC Qt${QT_VERSION_MAJOR}::Widgets
)

set_target_properties(${CUSTOMWIDGETS_TARGET_NAME} PROPERTIES
    PUBLIC_HEADER "colorpickercheckbox.h;${CMAKE_CURRENT_BINARY_DIR}/customWidgets_global.h"
)

install(TARGETS ${CUSTOMWIDGETS_TARGET_NAME}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    PUBLIC_HEADER DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/customWidgets
)
